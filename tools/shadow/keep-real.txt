# Classes kept as real (Forge-patched, SRG) bytecode instead of Rust-dispatched stubs. One regex per line,
# matched with find() against e.g. net/minecraft/client/gui/GuiScreen.class. Real classes call the stubbed
# ones (Minecraft, FontRenderer, Tessellator...) for everything Rust owns. Add a line to stop stubbing a class
# whose logic is pure Java; remove it to implement the class in Rust instead.
# --- GUI base classes mods subclass: vanilla widget logic, Forge's GuiScreenEvent post sites included
^net/minecraft/client/gui/(Gui|GuiScreen|GuiButton|GuiTextField|GuiSlot|GuiLabel)(\$[^/]*)?\.class$
# --- plain data
^net/minecraft/util/(Session|ChatAllowedCharacters|StringUtils|ResourceLocation|ChatComponent[A-Za-z]*|ChatStyle|ClickEvent|HoverEvent|IChatComponent)(\$[^/]*)?\.class$
^net/minecraft/util/(AxisAlignedBB|Vec3|Vec3i|BlockPos|EnumTypeAdapterFactory)(\$[^/]*)?\.class$
^net/minecraft/entity/DataWatcher(\$[^/]*)?\.class$
^net/minecraft/network/PacketBuffer\.class$
^net/minecraft/network/play/client/
# --- Forge events and the reflection helper (pure Java)

^net/minecraftforge/event/
^net/minecraftforge/fml/common/gameevent/

^net/minecraft/util/(EnumChatFormatting|EnumFacing|MovingObjectPosition|FoodStats|RegistryNamespaced|RegistrySimple|ObjectIntIdentityMap|IntHashMap|EnumParticleTypes|EnumWorldBlockLayer)(\$[^/]*)?\.class$
^net/minecraft/nbt/

# --- Forge config and progress bars: plain Java over files
^net/minecraftforge/common/config/

^net/minecraftforge/fml/relauncher/FMLInjectionData\.class$
^net/minecraftforge/fml/common/ModMetadata\.class$
# --- Block/Item registries: vanilla's own Bootstrap fills them (ModLoader.registries), so Blocks.*/Items.* and
#     states, properties, materials, tools and creative tabs are the real objects mods compare against
^net/minecraft/block/
^net/minecraft/init/(Blocks|Items)\.class$
^net/minecraft/item/
^net/minecraft/creativetab/
^net/minecraft/util/(RegistryNamespacedDefaultedByKey|RegistryDefaulted|IRegistry|IObjectIntIterable|MathHelper|Cartesian|MapPopulator|LongHashMap|JsonUtils|WeightedRandom|WeightedRandomChestContent|EnumWorldBlockLayer|EnumFacing|StatCollector)(\$[^/]*)?\.class$
^net/minecraftforge/fml/common/registry/(GameData|FMLControlledNamespacedRegistry|PersistentRegistryManager|RegistryDelegate|IncompatibleSubstitutionException|ExistingSubstitutionException|GameRegistry)(\$[^/]*)?\.class$
^net/minecraftforge/common/property/
# --- biomes: BiomeGenBase.getBiome over the vanilla list (Live.biome reads the mirrored chunk ids)
^net/minecraft/world/biome/(BiomeGenBase|BiomeGen[A-Za-z]*)(\$[^/]*)?\.class$
